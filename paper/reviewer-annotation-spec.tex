% ============================================================
% reviewer-annotation-spec.tex
% Cosmetic, review-only markup: highlights each reviewer comment's resolving
% passage in the manuscript (Reviewer 1 = orange, Reviewer 2 = blue) and prints
% the comment + resolution as a colour-coded footnote anchored at that passage.
%
% Why footnotes (not margin notes): MDPI's narrow margins and dense [H] floats
% make margin notes either overlap (marginnote, which does not float) or get
% dropped ("Float(s) lost", todonotes/marginpar). Footnotes never overlap, are
% fully legible, split across pages when long, sit on the same page as the
% highlighted text, and use a separate insert mechanism that does not strand
% the manuscript's figures and tables.
%
% This file is \input only by paper-annotated.tex. The clean manuscript
% (paper.tex + sections/*.tex) does not load it and is unchanged.
%
% To produce a clean (markup-free) build, load changes with [final]:
%   \usepackage[final]{changes}
% ============================================================

% Per-author change tracking; comments are rendered as footnotes below.
\usepackage[
  markup=underlined,
  authormarkup=none,
  commentmarkup=footnote
]{changes}

% Reviewers as "authors" with distinct colours
\definechangesauthor[name={Reviewer 1}, color=orange]{Rone}
\definechangesauthor[name={Reviewer 2}, color=blue]{Rtwo}

% Math/macro-safe in-text highlight: only recolour the span in the reviewer's
% colour (the per-author "authorcolor" is set by changes before the markup
% expands), so spans may safely contain inline math ($...$), \emph, \texttt,
% \ref, etc. without soul/ulem breakage.
\sethighlightmarkup{\textcolor{authorcolor}{#1}}
\setaddedmarkup{\textcolor{authorcolor}{#1}}

% Render the reviewer comment as a colour-coded footnote at the highlight.
\setcommentmarkup{\footnote{\textcolor{authorcolor}{\textbf{[Review note]}~#1}}}

% ===== Convenience macros (see docs/overleaf-comments.md) =====
% <comment> is a "Cxx: comment -> resolution" string shown in the footnote.

% --- Reviewer 1 (orange) ---
\newcommand{\RoneAdd}[2][]{\added[id=Rone, comment={#1}]{#2}}
\newcommand{\RoneDel}[2][]{\deleted[id=Rone, comment={#1}]{#2}}
\newcommand{\RoneRep}[3][]{\replaced[id=Rone, comment={#1}]{#3}{#2}}
\newcommand{\RoneHl}[2][]{\highlight[id=Rone, comment={#1}]{#2}}
\newcommand{\RoneCom}[1]{\comment[id=Rone]{#1}}

% --- Reviewer 2 (blue) ---
\newcommand{\RtwoAdd}[2][]{\added[id=Rtwo, comment={#1}]{#2}}
\newcommand{\RtwoDel}[2][]{\deleted[id=Rtwo, comment={#1}]{#2}}
\newcommand{\RtwoRep}[3][]{\replaced[id=Rtwo, comment={#1}]{#3}{#2}}
\newcommand{\RtwoHl}[2][]{\highlight[id=Rtwo, comment={#1}]{#2}}
\newcommand{\RtwoCom}[1]{\comment[id=Rtwo]{#1}}
