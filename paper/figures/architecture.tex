% figures/architecture.tex -- ACD System Architecture (journal-quality TikZ)
% Redrawn for clarity: orthogonal routing, no crossing arrows.  The three
% layers (attribution backend, POMDP agent, intervention engine) read top-down
% on each side; the two feedback paths are routed around the outer perimeter.
\begin{tikzpicture}[
    font=\small,
    box/.style={
        rectangle, draw=black!60, fill=blue!6,
        rounded corners=3pt, minimum width=3.0cm, minimum height=0.72cm,
        align=center, line width=0.5pt, font=\small
    },
    agentbox/.style={
        rectangle, draw=black!60, fill=red!5,
        rounded corners=3pt, minimum width=3.0cm, minimum height=0.72cm,
        align=center, line width=0.5pt, font=\small
    },
    matbox/.style={
        rectangle, draw=black!50, fill=gray!8,
        rounded corners=2pt, minimum width=1.55cm, minimum height=0.62cm,
        align=center, line width=0.4pt, font=\scriptsize
    },
    outbox/.style={
        rectangle, draw=green!50!black, fill=green!6,
        rounded corners=3pt, minimum width=3.0cm, minimum height=0.72cm,
        align=center, line width=0.5pt, font=\small
    },
    arr/.style={-{Latex[length=2mm]}, thick, draw=black!55},
    fb/.style={-{Latex[length=2mm]}, thick, draw=black!45, dashed},
    groupbox/.style={
        draw=black!35, dashed, fill=none, rounded corners=5pt, inner sep=7pt
    },
    grouplabel/.style={font=\scriptsize\bfseries, text=black!60},
]

% ===== Left column: Attribution Graph Backend (x = 0) =====
\node[box] (prompt)    at (0, 0)    {Prompt $x$};
\node[box] (llm)       at (0,-1.25) {LLM with Transcoders};
\node[box] (eap)       at (0,-2.50) {circuit-tracer / EAP};
\node[box] (attrgraph) at (0,-3.75) {Attribution Graph\\$\{(\ell,p,f)\},\ \mathbf{W},\ \mathbf{a}$};
\node[box] (prune)     at (0,-5.00) {Pruning ($\geq$80\% influence)};
\node[box] (cand)      at (0,-6.25) {Candidate Features + $\mathrm{imp}(i)$};

\begin{scope}[on background layer]
  \node[groupbox, fit=(prompt)(llm)(eap)(attrgraph)(prune)(cand),
        label={[grouplabel, yshift=2pt]above:Attribution Graph Backend}] (backendgrp) {};
\end{scope}

% ===== Right column: POMDP Agent (x = 6.6) =====
% Generative model grouped into a single container (one arrow out).
\node[matbox] (Amat) at (5.85, 0.00) {$\mathbf{A}$: $P(o|s)$};
\node[matbox] (Bmat) at (7.55, 0.00) {$\mathbf{B}(a)$: $P(s'|s,a)$};
\node[matbox] (Cmat) at (5.85,-0.78) {$\mathbf{C}$: $P(o)$};
\node[matbox] (Dmat) at (7.55,-0.78) {$\mathbf{D}$: $P(s)$};
\begin{scope}[on background layer]
  \node[groupbox, draw=red!35, fit=(Amat)(Bmat)(Cmat)(Dmat),
        label={[grouplabel, text=red!50!black, yshift=1pt]above:Generative Model}] (genmodel) {};
\end{scope}

\node[agentbox] (infer)  at (6.70,-2.30) {Variational Inference $q(s|o)$};
\node[agentbox] (efe)    at (6.70,-3.55) {EFE $G(i,a)$: epistemic\,+\,pragmatic};
\node[agentbox] (select) at (6.70,-4.80) {Select $(i^*,a^*)=\arg\min G$};

\begin{scope}[on background layer]
  \node[groupbox, draw=red!30, fit=(genmodel)(infer)(efe)(select),
        label={[grouplabel, text=red!50!black]above:Active Inference POMDP Agent}] (agentgrp) {};
\end{scope}

% ===== Bottom: Intervention engine (centred under agent) =====
\node[outbox] (intervene) at (6.70,-6.25) {\texttt{feature\_intervention}\\(ablation / patching / steering)};
\node[outbox] (obs)       at (6.70,-7.55) {KL divergence + observation $o$};

% ===== Backend vertical flow =====
\draw[arr] (prompt)    -- (llm);
\draw[arr] (llm)       -- (eap);
\draw[arr] (eap)       -- (attrgraph);
\draw[arr] (attrgraph) -- (prune);
\draw[arr] (prune)     -- (cand);

% ===== Candidates -> agent (clean elbow, lane at x=2.7) =====
\draw[arr] (cand.east) -| (2.7,-2.30) -- (infer.west)
    node[pos=0.72, above, font=\scriptsize]{candidates};

% ===== Generative model -> inference (single arrow) =====
\draw[arr] (genmodel.south) -- (infer.north);

% ===== Agent pipeline =====
\draw[arr] (infer)  -- (efe);
\draw[arr] (efe)    -- (select);
\draw[arr] (select) -- (intervene);
\draw[arr] (intervene) -- (obs);

% ===== Feedback 1: intervention spec -> LLM (perimeter, left side) =====
% Down to a bottom lane, left around the backend, up the far-left edge to the LLM.
\draw[fb] (intervene.west) -| (4.05,-7.25) -- (-2.40,-7.25) |- (llm.west)
    node[pos=0.18, below, font=\scriptsize]{$(\ell,p,f,\mathrm{val})$};

% ===== Feedback 2: observation -> Dirichlet update of generative model =====
% Right around the agent, up the far-right edge to the generative-model block.
\draw[fb] (obs.east) -| (8.95,0.0) -- (Bmat.east)
    node[pos=0.16, right, font=\scriptsize]{Dirichlet update};

\end{tikzpicture}
