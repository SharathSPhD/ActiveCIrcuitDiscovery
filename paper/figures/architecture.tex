\begin{tikzpicture}[
    font=\small,
    node distance=0.6cm and 1.0cm,
    box/.style={
        rectangle, draw=black!60, fill=blue!6,
        rounded corners=3pt, minimum width=3.0cm, minimum height=0.7cm,
        align=center, line width=0.5pt, font=\small
    },
    agentbox/.style={
        rectangle, draw=black!60, fill=red!5,
        rounded corners=3pt, minimum width=2.6cm, minimum height=0.7cm,
        align=center, line width=0.5pt, font=\small
    },
    matbox/.style={
        rectangle, draw=black!50, fill=gray!8,
        rounded corners=2pt, minimum width=1.8cm, minimum height=0.6cm,
        align=center, line width=0.4pt, font=\scriptsize
    },
    outbox/.style={
        rectangle, draw=green!50!black, fill=green!5,
        rounded corners=3pt, minimum width=2.4cm, minimum height=0.7cm,
        align=center, line width=0.5pt, font=\small
    },
    arr/.style={-{Latex[length=2mm]}, thick, draw=black!55},
    dasharr/.style={-{Latex[length=2mm]}, thick, draw=black!40, dashed},
    groupbox/.style={
        draw=black!35, dashed, fill=none, rounded corners=5pt, inner sep=6pt
    },
    grouplabel/.style={font=\scriptsize\bfseries, text=black!60},
]

% ---- Left: Attribution Graph Backend ----
\node[box] (prompt) {Prompt $x$};
\node[box, below=0.5cm of prompt] (llm) {LLM with Transcoders};
\node[box, below=0.5cm of llm] (eap) {circuit-tracer / EAP};
\node[box, below=0.5cm of eap] (attrgraph) {Attribution Graph\\$\{(\ell,p,f)\}$, $\mathbf{W}$, $\mathbf{a}$};
\node[box, below=0.5cm of attrgraph] (prune) {Pruning ($\geq$80\% influence)};
\node[box, below=0.5cm of prune] (cand) {Candidate Features\\+ $\mathrm{imp}(i)$};

\begin{scope}[on background layer]
  \node[groupbox, fit=(prompt)(llm)(eap)(attrgraph)(prune)(cand),
        label={[grouplabel, yshift=2pt]above:Attribution Graph Backend}] (backendgrp) {};
\end{scope}

% ---- Right: POMDP Agent ----
\node[matbox, right=3.5cm of llm] (Amat) {$\mathbf{A}$: Obs.\ model\\$A(o|s)$};
\node[matbox, right=0.3cm of Amat] (Bmat) {$\mathbf{B}(a)$: Trans.\\$B(s'|s,a)$};
\node[matbox, below=0.4cm of Amat] (Cmat) {$\mathbf{C}$: Preference\\$C(o)$};
\node[matbox, below=0.4cm of Bmat] (Dmat) {$\mathbf{D}$: Prior\\$D(s)$};

\node[agentbox, below=0.6cm of Cmat, xshift=0.5cm] (infer) {Variational Inference\\$q(s|o)$};
\node[agentbox, below=0.5cm of infer] (efe) {EFE: $G(i,a)$\\epistemic + pragmatic};
\node[agentbox, below=0.5cm of efe] (select) {Select $(i^*,a^*)$\\$= \arg\min G$};

\begin{scope}[on background layer]
  \node[groupbox, draw=red!30, fit=(Amat)(Bmat)(Cmat)(Dmat)(infer)(efe)(select),
        label={[grouplabel, text=red!50!black]above:Active Inference POMDP Agent}] (agentgrp) {};
\end{scope}

% ---- Bottom: Intervention and Feedback ----
\node[outbox, below=0.6cm of select] (intervene) {feature\_intervention\\(ablation / patching / steering)};
\node[outbox, below=0.5cm of intervene] (obs) {KL Divergence\\+ Updated Observations};

% ---- Arrows: Backend flow ----
\draw[arr] (prompt) -- (llm);
\draw[arr] (llm) -- (eap);
\draw[arr] (eap) -- (attrgraph);
\draw[arr] (attrgraph) -- (prune);
\draw[arr] (prune) -- (cand);

% ---- Arrow: Candidates to Agent ----
\draw[arr] (cand.east) -- ++(1.0,0) |- (infer.west)
    node[pos=0.55, above, font=\scriptsize]{candidates};

% ---- Arrows: Generative model to inference/EFE ----
\draw[arr] (Amat.south) -- ++(0,-0.15) -| ([xshift=-0.3cm]infer.north);
\draw[arr] (Dmat.south) -- ++(0,-0.15) -| ([xshift=0.3cm]infer.north);
\draw[arr] (Bmat.south) -- ++(0,-0.55) -| ([xshift=-0.3cm]efe.north);
\draw[arr] (Cmat.south) -| ([xshift=0.3cm]efe.north);

% ---- Arrows: Agent pipeline ----
\draw[arr] (infer) -- (efe);
\draw[arr] (efe) -- (select);
\draw[arr] (select) -- (intervene);

% ---- Arrow: Intervention to LLM ----
\draw[dasharr] (intervene.west) -- ++(-2.2,0) |- (llm.west)
    node[pos=0.08, above, font=\scriptsize]{$(l,p,f,\mathrm{val})$};

% ---- Arrow: Observations feedback ----
\draw[arr] (intervene) -- (obs);
\draw[dasharr] (obs.east) -- ++(1.4,0) |- (Bmat.east)
    node[pos=0.12, right, font=\scriptsize]{Dirichlet update};
