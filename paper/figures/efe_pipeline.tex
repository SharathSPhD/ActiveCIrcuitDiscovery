% figures/efe_pipeline.tex -- POMDP Agent Pipeline (journal-quality TikZ)
\begin{tikzpicture}[
    font=\small,
    node distance=0.5cm and 0.7cm,
    mat/.style={
        rectangle, draw=blue!50!black, fill=blue!6,
        rounded corners=3pt, minimum width=1.8cm, minimum height=0.7cm,
        align=center, line width=0.5pt, font=\small\bfseries
    },
    proc/.style={
        rectangle, draw=red!40!black, fill=red!5,
        rounded corners=3pt, minimum width=2.4cm, minimum height=0.7cm,
        align=center, line width=0.5pt, font=\small
    },
    io/.style={
        rectangle, draw=green!40!black, fill=green!5,
        rounded corners=3pt, minimum width=2.2cm, minimum height=0.7cm,
        align=center, line width=0.5pt, font=\small
    },
    sel/.style={
        rectangle, draw=orange!60!black, fill=orange!6,
        rounded corners=3pt, minimum width=2.8cm, minimum height=0.7cm,
        align=center, line width=0.5pt, font=\small
    },
    arr/.style={-{Latex[length=2mm]}, thick, draw=black!50},
    looparr/.style={-{Latex[length=2mm]}, thick, draw=black!35, dashed},
]

% Row 1: Generative model matrices
\node[mat] (A) {$\mathbf{A}$\\likelihood};
\node[mat, right=0.5cm of A] (B) {$\mathbf{B}(a)$\\transition};
\node[mat, right=0.5cm of B] (C) {$\mathbf{C}$\\preference};
\node[mat, right=0.5cm of C] (D) {$\mathbf{D}$\\prior};

% Row 2: Inference and EFE
\node[proc, below=0.8cm of A, xshift=0.7cm] (infer) {Infer states\\$q(s \mid o)$};
\node[proc, right=0.8cm of infer] (efe) {EFE: $G(i,a)$\\epistemic + pragmatic};

% Row 3: Selection
\node[sel, below=0.7cm of infer, xshift=2.0cm] (select) {$(i^*,a^*) = \arg\min G$};

% Row 4: Execute and observe
\node[io, below left=0.7cm and 0.3cm of select] (exec) {Execute $a^*$\\on feature $i^*$};
\node[io, below right=0.7cm and 0.3cm of select] (obs) {Observe\\KL, act., conn.};

% Row 5: Online learning
\node[io, below=0.6cm of select, yshift=-1.2cm] (learn) {Online Learning\\Dirichlet update $\mathbf{A}$};

% Arrows: matrices -> inference
\draw[arr] (A.south) -- ++(0,-0.2) -| ([xshift=-0.3cm]infer.north);
\draw[arr] (D.south) -- ++(0,-0.2) -| ([xshift=0.3cm]infer.north);

% Arrows: matrices -> EFE
\draw[arr] (B.south) -- ++(0,-0.2) -| ([xshift=-0.3cm]efe.north);
\draw[arr] (C.south) -- ++(0,-0.2) -| ([xshift=0.3cm]efe.north);

% Inference -> EFE -> Selection
\draw[arr] (infer) -- (efe);
\draw[arr] (efe.south) -- ++(0,-0.2) -| (select.north);

% Selection -> execute and observe
\draw[arr] (select.south) -- ++(0,-0.2) -| (exec.north);
\draw[arr] (select.south) -- ++(0,-0.2) -| (obs.north);

% Execute and observe -> learning
\draw[arr] (exec.south) -- ++(0,-0.25) -| (learn.west);
\draw[arr] (obs.south) -- ++(0,-0.25) -| (learn.east);

% Learning -> back to generative model (loop)
\draw[looparr] (learn.west) -- ++(-1.6,0) |- (A.west)
    node[pos=0.3, left, font=\scriptsize]{update $\mathbf{A}$, iterate};

\end{tikzpicture}
